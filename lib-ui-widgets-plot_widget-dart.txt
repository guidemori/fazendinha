import 'package:flutter/material.dart';

import '../../models/plot.dart';

/// Representa visualmente um canteiro na fazenda.
class PlotWidget extends StatelessWidget {
  const PlotWidget({super.key, required this.plot});

  final Plot plot;

  @override
  Widget build(BuildContext context) {
    final isEmpty = plot.isEmpty;

    return Container(
      decoration: BoxDecoration(
        color: isEmpty ? const Color(0xFF8D6E63) : const Color(0xFF66BB6A),
        borderRadius: BorderRadius.circular(16),
        border: Border.all(color: const Color(0xFF5D4037), width: 3),
      ),
      alignment: Alignment.center,
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          Text(
            plot.isEmpty ? '🟫' : (plot.plantedCrop?.emoji ?? '🌱'),
            style: const TextStyle(fontSize: 40),
          ),
          const SizedBox(height: 8),
          Text(
            plot.isEmpty ? 'Canteiro vazio' : 'Crescendo...',
            style: const TextStyle(
              color: Colors.white,
              fontWeight: FontWeight.bold,
            ),
          ),
        ],
      ),
    );
  }
}
